    const rx = reactionsFor(p.id, DB.meId);
    return { id: p.id, author: uAuthor(p.author_id), body: p.body, photo: false, created: p.created,
      likes: rx.total, liked: !!rx.mine, myReaction: rx.mine, reactBy: rx.by,
      comments: DB.comments.filter((c) => c.post_id === p.id).length };
  }
  function friendIdsOf(id) { const s = []; DB.friends.forEach((k) => { const [a, b] = k.split(&quot;-&quot;).map(Number); if (a === id) s.push(b); else if (b === id) s.push(a); }); return s; }

  // Route a request against the in-browser store. Returns the same shapes as the real API.
  function route(path, method, body) {
    const seg = path.split(&quot;?&quot;)[0].split(&quot;/&quot;).filter(Boolean); // e.g. [&quot;api&quot;,&quot;posts&quot;,&quot;3&quot;,&quot;like&quot;]
    const qs = new URLSearchParams(path.split(&quot;?&quot;)[1] || &quot;&quot;);
    const me = DB.meId;
    if (path === &quot;/api/me&quot;) { const p = DB.users[me]; return { user: { id: me, email: &quot;you@demo&quot; }, profile: { name: p.name, bio: p.bio, photo: false, cover: false, support: { on: !!p.need_support, note: p.support_note } } }; }
    if (path === &quot;/api/logout&quot; || path === &quot;/api/notifications/read&quot;) return { ok: true };
    if (path === &quot;/api/feed&quot;) {
      const ids = [me, ...friendIdsOf(me)];
      return { posts: DB.posts.filter((p) => ids.includes(p.author_id)).sort((a, b) => b.created - a.created).map(postView) };
    }
    if (path === &quot;/api/posts&quot; &amp;&amp; method === &quot;POST&quot;) {
      const id = ++DB.seq.post; DB.posts.push({ id, author_id: me, body: String(body.body || &quot;&quot;).trim(), created: Date.now() });
      return { ok: true, id };
    }
    let m;
    if ((m = path.match(/^\/api\/posts\/(\d+)\/like$/)) &amp;&amp; method === &quot;POST&quot;) {
      const pid = +m[1]; DB.likes[pid] = DB.likes[pid] || {};
      const type = REACTIONS.includes(body.type) ? body.type : &quot;like&quot;;
      if (DB.likes[pid][me] === type) delete DB.likes[pid][me]; else DB.likes[pid][me] = type;
      const rx = reactionsFor(pid, me); return { ok: true, likes: rx.total, liked: !!rx.mine, myReaction: rx.mine, reactBy: rx.by };
    }
    if ((m = path.match(/^\/api\/posts\/(\d+)\/comments$/)) &amp;&amp; method === &quot;GET&quot;) {
      const pid = +m[1]; return { comments: DB.comments.filter((c) => c.post_id === pid).map((c) => ({ author: uAuthor(c.user_id), body: c.body, created: c.created })) };
    }
    if ((m = path.match(/^\/api\/posts\/(\d+)\/comments$/)) &amp;&amp; method === &quot;POST&quot;) {
      const pid = +m[1]; DB.comments.push({ id: ++DB.seq.comment, post_id: pid, user_id: me, body: String(body.body || &quot;&quot;).trim(), created: Date.now() });
      return { ok: true, comments: DB.comments.filter((c) => c.post_id === pid).length };
    }
    if (path.startsWith(&quot;/api/people&quot;)) {
      const term = (qs.get(&quot;q&quot;) || &quot;&quot;).toLowerCase();
      const ids = Object.keys(DB.users).map(Number).filter((id) => id !== me &amp;&amp; (!term || DB.users[id].name.toLowerCase().includes(term)));
      return { people: ids.map(personOf) };
    }
    if (path === &quot;/api/friends&quot;) return { friends: friendIdsOf(me).map(personOf) };
    if (path === &quot;/api/friends/requests&quot;) return { requests: DB.pending.filter((p) => p.to === me).map((p) => personOf(p.from)) };
    if (path === &quot;/api/friends/request&quot; &amp;&amp; method === &quot;POST&quot;) {
      const t = +body.toId;
      if (DB.pending.some((p) => p.from === t &amp;&amp; p.to === me)) { DB.pending = DB.pending.filter((p) => !(p.from === t &amp;&amp; p.to === me)); DB.friends.add(pairKey(me, t)); return { ok: true, rel: &quot;friends&quot; }; }
      DB.pending.push({ from: me, to: t, created: Date.now() }); return { ok: true, rel: &quot;outgoing&quot; };
    }
    if (path === &quot;/api/friends/remove&quot; &amp;&amp; method === &quot;POST&quot;) {
      const t = +body.userId; DB.friends.delete(pairKey(me, t)); DB.pending = DB.pending.filter((p) => !((p.from === me &amp;&amp; p.to === t) || (p.from === t &amp;&amp; p.to === me))); return { ok: true };
    }
    if ((m = path.match(/^\/api\/users\/(\d+)$/)) &amp;&amp; method === &quot;GET&quot;) {
      const id = +m[1]; const p = DB.users[id]; if (!p) throw new Error(&quot;No such person.&quot;);
      const rel = relOf(id); const canSee = id === me || rel === &quot;friends&quot;;
      return { user: { id, name: p.name, bio: p.bio, photo: false, cover: false, rel, isMe: id === me, friendCount: friendIdsOf(id).length, postCount: DB.posts.filter((x) => x.author_id === id).length, support: canSee &amp;&amp; p.need_support ? { on: true, note: p.support_note } : { on: false } },
        posts: DB.posts.filter((x) => x.author_id === id).sort((a, b) => b.created - a.created).map(postView) };
    }
    if (path === &quot;/api/profile&quot; &amp;&amp; method === &quot;PUT&quot;) { const p = DB.users[me]; p.name = String(body.name || &quot;&quot;).slice(0, 60); p.bio = String(body.bio || &quot;&quot;).slice(0, 300); return { ok: true, profile: { name: p.name, bio: p.bio, photo: false, cover: false } }; }
    if (path === &quot;/api/profile/photo&quot; || path === &quot;/api/profile/cover&quot;) return { ok: true };
    if (path === &quot;/api/support&quot; &amp;&amp; method === &quot;PUT&quot;) { const p = DB.users[me]; p.need_support = body.on ? 1 : 0; p.support_note = body.on ? String(body.note || &quot;&quot;).slice(0, 200) : &quot;&quot;; return { ok: true, support: { on: !!p.need_support, note: p.support_note } }; }
    if (path === &quot;/api/support/circle&quot;) { return { circle: friendIdsOf(me).map((id) => DB.users[id]).filter((u) => u.need_support).map((u) => ({ id: u.id, name: u.name, photo: false, note: u.support_note, since: mins(200) })) }; }
    if (path === &quot;/api/support/reach&quot; &amp;&amp; method === &quot;POST&quot;) return { ok: true };
    if (path === &quot;/api/dm/threads&quot;) {
      const partners = [...new Set(DB.dms.filter((d) => d.from === me || d.to === me).map((d) => (d.from === me ? d.to : d.from)))];
      return { threads: partners.map((pid) => { const msgs = DB.dms.filter((d) => (d.from === me &amp;&amp; d.to === pid) || (d.from === pid &amp;&amp; d.to === me)); const last = msgs[msgs.length - 1]; return { with: uAuthor(pid), last: last ? { body: last.body, mine: last.from === me, created: last.created } : null, unread: msgs.filter((d) => d.to === me &amp;&amp; !d.read).length }; }) };
    }
    if (path === &quot;/api/dm/unread&quot;) return { unread: DB.dms.filter((d) => d.to === me &amp;&amp; !d.read).length };
    if (path.startsWith(&quot;/api/dm&quot;) &amp;&amp; method === &quot;GET&quot;) { const pid = +qs.get(&quot;with&quot;); DB.dms.forEach((d) => { if (d.from === pid &amp;&amp; d.to === me) d.read = true; }); return { with: uAuthor(pid), messages: DB.dms.filter((d) => (d.from === me &amp;&amp; d.to === pid) || (d.from === pid &amp;&amp; d.to === me)).map((d) => ({ mine: d.from === me, body: d.body, created: d.created })) }; }
    if (path === &quot;/api/dm&quot; &amp;&amp; method === &quot;POST&quot;) { DB.dms.push({ from: me, to: +body.toId, body: String(body.body || &quot;&quot;).trim(), created: Date.now(), read: true }); return { ok: true }; }
    if (path === &quot;/api/groups&quot; &amp;&amp; method === &quot;GET&quot;) return { groups: DB.groups.map((g) => ({ id: g.id, name: g.name, about: g.about, members: g.members.length, mine: g.members.includes(me) })) };
    if (path === &quot;/api/groups&quot; &amp;&amp; method === &quot;POST&quot;) { const id = ++DB.seq.group; DB.groups.push({ id, name: String(body.name || &quot;Group&quot;).slice(0, 60), about: String(body.about || &quot;&quot;).slice(0, 300), owner_id: me, members: [me] }); return { ok: true, id }; }
    if (path === &quot;/api/notifications&quot; &amp;&amp; method === &quot;GET&quot;) { return { unread: DB.notifs.filter((n) => !n.read).length, notifications: DB.notifs.map((n) => ({ id: n.id, type: n.type, actor: uAuthor(n.actor_id), postId: n.post_id, created: n.created, read: !!n.read })) }; }
    return { ok: true };
  }
  async function api(path, opts = {}) {
    await new Promise((r) => setTimeout(r, 60)); // tiny delay so it feels real
    let body = {}; try { body = opts.body ? (typeof opts.body === &quot;string&quot; ? JSON.parse(opts.body) : opts.body) : {}; } catch {}
    return route(path, opts.method || &quot;GET&quot;, body);
  }

  function esc(s) { return String(s == null ? &quot;&quot; : s).replace(/[&amp;<>&quot;']/g, (c) => ({ &quot;&amp;&quot;: &quot;&amp;amp;&quot;, &quot;<&quot;: &quot;&amp;lt;&quot;, &quot;>&quot;: &quot;&amp;gt;&quot;, '&quot;': &quot;&amp;quot;&quot;, &quot;'&quot;: &quot;&amp;#39;&quot; }[c])); }
  function initials(n) { return (n || &quot;?&quot;).trim().split(/\s+/).map((w) => w[0]).slice(0, 2).join(&quot;&quot;).toUpperCase(); }
  function timeago(ts) {
    const s = Math.floor((Date.now() - ts) / 1000);
    if (s < 60) return &quot;just now&quot;;
    const m = Math.floor(s / 60); if (m < 60) return m + &quot;m&quot;;
    const h = Math.floor(m / 60); if (h < 24) return h + &quot;h&quot;;
    const d = Math.floor(h / 24); if (d < 7) return d + &quot;d&quot;;
    return new Date(ts).toLocaleDateString();
  }

          const box = card.querySelector(&quot;.comments&quot;);
          if (box &amp;&amp; !box.hidden) { const form = box.querySelector(&quot;.c-form&quot;); const node = renderComment(d.comment); form ? form.before(node) : box.appendChild(node); }
        }
      });
    });
    es.addEventListener(&quot;new_post&quot;, (ev) => {
      let p; try { p = JSON.parse(ev.data); } catch { return; }
      if (el(&quot;view-feed&quot;).hidden) return;
      el(&quot;feedEmpty&quot;).hidden = true;
      el(&quot;posts&quot;).prepend(renderPost(p));
    });
    es.onerror = () => {};
  }
  function closeStream() { if (es) { es.close(); es = null; } }

  /* ---------- Messages ---------- */
  let currentDm = null;
  function showThreadList() { el(&quot;threadListCard&quot;).hidden = false; el(&quot;chatCard&quot;).hidden = true; currentDm = null; }
  async function loadThreads() {
    let d; try { d = await api(&quot;/api/dm/threads&quot;); } catch { return; }
    const list = el(&quot;threadList&quot;); list.innerHTML = &quot;&quot;;
    el(&quot;threadEmpty&quot;).hidden = d.threads.length > 0;
    d.threads.forEach((t) => {
      const av = avatarInner(t.with);
      const row = document.createElement(&quot;button&quot;); row.className = &quot;thread&quot;;
      const preview = t.last ? (t.last.mine ? &quot;You: &quot; : &quot;&quot;) + t.last.body : &quot;Say hi&quot;;
      row.innerHTML = `<div class=&quot;avatar&quot; ${av.style ? `style=&quot;${av.style}&quot;` : &quot;&quot;}>${av.text}</div>
        <div class=&quot;who&quot;><div class=&quot;t-name&quot;>${esc(t.with.name)}</div><div class=&quot;t-last ${t.unread ? &quot;unread&quot; : &quot;&quot;}&quot;>${esc(preview)}</div></div>
        ${t.unread ? '<span class=&quot;dot&quot;></span>' : &quot;&quot;}`;
      row.onclick = () => openThread(t.with.id, t.with.name);
      list.appendChild(row);
    });
  }
  async function openThread(id, name) {
    currentDm = id; el(&quot;chatWith&quot;).textContent = name;
    el(&quot;threadListCard&quot;).hidden = true; el(&quot;chatCard&quot;).hidden = false;
    show(&quot;messages&quot;);
    await renderThread(); refreshDmBadge();
  }
  async function renderThread() {
    const log = el(&quot;chatLog&quot;);
    let d; try { d = await api(&quot;/api/dm?with=&quot; + currentDm); } catch { return; }
    el(&quot;chatWith&quot;).textContent = d.with.name;
    log.innerHTML = &quot;&quot;;
    if (!d.messages.length) log.innerHTML = '<p class=&quot;chat-empty&quot;>No messages yet. Send the first one.</p>';
    d.messages.forEach((m) => { const b = document.createElement(&quot;div&quot;); b.className = &quot;msg &quot; + (m.mine ? &quot;me&quot; : &quot;them&quot;); b.textContent = m.body; log.appendChild(b); });
    log.scrollTop = log.scrollHeight;
  }
  el(&quot;chatBack&quot;).onclick = () => { showThreadList(); loadThreads(); };
  el(&quot;chatForm&quot;).onsubmit = async (e) => {
    e.preventDefault();
    const input = el(&quot;chatInput&quot;); const body = input.value.trim(); if (!body || !currentDm) return; input.value = &quot;&quot;;
    const log = el(&quot;chatLog&quot;); const empty = log.querySelector(&quot;.chat-empty&quot;); if (empty) empty.remove();
    const b = document.createElement(&quot;div&quot;); b.className = &quot;msg me&quot;; b.textContent = body; log.appendChild(b); log.scrollTop = log.scrollHeight;
    try { await api(&quot;/api/dm&quot;, { method: &quot;POST&quot;, body: { toId: currentDm, body } }); } catch (err) { toast(err.message); }
  };
  function setDmBadge(n) { const b = el(&quot;dmBadge&quot;); if (n > 0) { b.textContent = n; b.hidden = false; } else b.hidden = true; }
  async function refreshDmBadge() { try { setDmBadge((await api(&quot;/api/dm/unread&quot;)).unread); } catch {} }

  /* ---------- Boot ---------- */
  async function boot() {
    try {
      me = await api(&quot;/api/me&quot;);
      el(&quot;topbar&quot;).hidden = false;
      paintMe();
      if (!me.profile.name) { show(&quot;profile&quot;); loadProfile(); return; }
      refreshReqBadge();
      refreshNotifBadge();
      refreshDmBadge();
      openStream();
      go(&quot;feed&quot;);
      setTimeout(() => toast(&quot;Demo — explore freely. Your changes stay in this browser.&quot;), 900);
    } catch { me = null; el(&quot;topbar&quot;).hidden = true; setMode(&quot;signup&quot;); show(&quot;auth&quot;); }
  }
  boot();
})();
